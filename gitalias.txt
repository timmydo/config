[alias]
    c = commit
    co = checkout
    b = branch
    s = status
    p = push
    l = log --graph --oneline
    cof = !git for-each-ref --format='%(refname:short)' refs/heads | fzf | xargs git checkout
    pom = pull origin master
    cam = commit -a -m fix
    rbi = rebase --interactive @{upstream}
    log-graph = log --graph --all  --decorate --oneline
    log-my-week = !git log --author $(git config user.email) --since "1 week ago"
    uncommit = reset --soft HEAD~1
    cleanout = !git clean -df && git checkout -- .
    ours   = !"f() { git checkout --ours $@ && git add $@; }; f"
    theirs = !"f() { git checkout --theirs $@ && git add $@; }; f"
    publish = "!git push -u origin $(git branch-name)"
    unpublish = "!git push origin :$(git branch-name)"    
    branches = branch -a
    stashes = stash list
    tags = tag -n1 --list
    squash-in = !"f() { test -n \"$1\" || { echo 'usage: git squash-in <branch>' >&2; return 2; }; test -z \"$(git status --porcelain)\" || { echo 'refusing: working tree not clean' >&2; return 1; }; base=$(git merge-base HEAD \"$1\") || { echo \"no merge base with $1\" >&2; return 1; }; echo 'Commits to squash:'; git --no-pager log --oneline --reverse \"$base\"..\"$1\"; git merge --squash \"$1\" || echo 'merge --squash hit conflicts — resolve (git status), or bail: git reset --hard HEAD' >&2; git log --reverse --format=%B \"$base\"..\"$1\" > \"$(git rev-parse --git-dir)/SQUASH_MSG\"; echo; echo 'Message prefilled from branch commits. Review: git diff --cached | Commit: git commit | Bail: git reset --hard HEAD'; }; f"
